# STEP 3 — Build Question Generator

Open:

```
src/core/sessionEngine.ts
```

Add:

```ts
import { concepts } from "../grammar/concepts"

export function generateQuestion() {

  const concept =
    concepts[Math.floor(Math.random() * concepts.length)]

  const example =
    concept.examples[
      Math.floor(Math.random() * concept.examples.length)
    ]

  return {
    concept: concept.name,
    question: "Translate to English",
    french: example
  }
}
```

# STEP 4 — Build Trainer UI

Open:

```
src/app/page.tsx
```

Replace everything with:

```tsx
"use client"

import { useState } from "react"
import { generateQuestion } from "../core/sessionEngine"

export default function TrainerPage() {

  const [question, setQuestion] = useState(generateQuestion())
  const [answer, setAnswer] = useState("")
  const [feedback, setFeedback] = useState("")

  function checkAnswer() {

    if (answer.length === 0) {
      setFeedback("Enter an answer.")
      return
    }

    setFeedback("Answer recorded.")

    setTimeout(() => {
      setQuestion(generateQuestion())
      setAnswer("")
      setFeedback("")
    }, 1500)
  }

  return (
    <main style={{ padding: 40 }}>

      <h1>French Military Trainer</h1>

      <h2>{question.question}</h2>

      <h3>{question.french}</h3>

      <input
        value={answer}
        onChange={(e) => setAnswer(e.target.value)}
        placeholder="Your answer"
      />

      <br />
      <br />

      <button onClick={checkAnswer}>
        Submit
      </button>

      <p>{feedback}</p>

    </main>
  )
}
```

```ts
export function generateQuestion(level = 1) {

  const filtered = concepts.filter(
    c => c.difficulty <= level
  )

  const concept =
    filtered[Math.floor(Math.random() * filtered.length)]

  const example =
    concept.examples[
      Math.floor(Math.random() * concept.examples.length)
    ]

  return {
    concept: concept.name,

    french: example

}
```
