# 🧱 STEP 7 — Create Core Engine Placeholder

`/src/core/sessionEngine.ts`

```ts
import { ConceptState } from "@/types/concept";

export function selectNextConcept(
  concepts: ConceptState[]
): ConceptState | null {
  if (!concepts.length) return null;

  return concepts.sort((a, b) => a.strength - b.strength)[0];
}
```

No logic perfection.
Just structure.

# 🧱 STEP 8 — Wire Engine to UI

Modify `/src/app/page.tsx`

```tsx
import { initialConcepts } from "@/grammar/concepts";
import { selectNextConcept } from "@/core/sessionEngine";

export default function Home() {
  const concept = selectNextConcept(initialConcepts);

  return (
    <main className="p-8">
      <h1 className="text-xl font-bold">French Military Trainer</h1>
      <p>Next Concept: {concept?.id}</p>
    </main>
  );
}
```
