# STEP 1 — Build Answer Evaluation Engine

Create:

```
src/core/answerEvaluator.ts
```

Code:

```ts
export function evaluateAnswer(
  userAnswer: string,
  correctAnswers: string[]
) {

  const normalizedUser =
    userAnswer.trim().toLowerCase()

  const normalizedCorrect =
    correctAnswers.map(a => a.toLowerCase())

  if (normalizedCorrect.includes(normalizedUser)) {
    return {
      correct: true,
      message: "Correct!"
    }
  }

  return {
    correct: false,
    message: "Not correct"
  }
}
```

# STEP 6 — Grammar Explanation Engine

Create:

```
src/core/grammarExplainer.ts
```

Code:

```ts
export function explainPhrase(french: string){

  const dictionary = {
    "je": "I",
    "suis": "am",
    "tu": "you",
    "es": "are",
    "nous": "we",
    "sommes": "are"
  }

  const words = french.split(" ")

  const explanation = words.map(w => ({
      word: w,
      meaning: dictionary[w] || "?"
  }))

  return explanation
}
```

Example output:

```
nous sommes

nous → we
sommes → are
```
