# STEP 2 — Implement Spaced Repetition Algorithm

Create file:

```
src/core/spacedRepetition.ts
```

```ts
import { ConceptProgress } from "../types/progress"

export function updateProgress(
  progress: ConceptProgress,
  quality: number
): ConceptProgress {

  if (quality < 3) {
    progress.repetitions = 0
    progress.interval = 1
  } else {

    progress.repetitions += 1

    if (progress.repetitions === 1) {
      progress.interval = 1
    } else if (progress.repetitions === 2) {
      progress.interval = 3
    } else {
      progress.interval =
        Math.round(progress.interval * progress.easeFactor)
    }

    progress.easeFactor =
      progress.easeFactor +
      (0.1 - (5 - quality) * (0.08 + (5 - quality) * 0.02))

    if (progress.easeFactor < 1.3) {
      progress.easeFactor = 1.3
    }

  }

  progress.nextReview =
    Date.now() + progress.interval * 24 * 60 * 60 * 1000

  return progress
}
```

This is a simplified **SuperMemo algorithm** used by flashcard systems.

# STEP 3 — Create Progress Storage

Create:

```
src/storage/progressStore.ts
```

```ts
import { ConceptProgress } from "../types/progress"

const KEY = "french_trainer_progress"

export function loadProgress(): ConceptProgress[] {

  if (typeof window === "undefined") return []

  const raw = localStorage.getItem(KEY)

  if (!raw) return []

  return JSON.parse(raw)
}

export function saveProgress(progress: ConceptProgress[]) {

  localStorage.setItem(KEY, JSON.stringify(progress))

}
```

For now we store progress in **localStorage**.

Later we upgrade to **Firebase**.

# STEP 4 — Add Concept Review Selection

Create in:

```
src/core/sessionEngine.ts
```

Add function:

```ts
import { loadProgress } from "../storage/progressStore"
import { grammarConcepts } from "../grammar/concepts"

export function selectNextConcept() {

  const progress = loadProgress()

  const now = Date.now()

  const dueConcepts = progress.filter(
    p => p.nextReview <= now
  )

  if (dueConcepts.length > 0) {

    const concept = dueConcepts[0]

    return grammarConcepts.find(
      g => g.id === concept.conceptId
    )

  }

  return grammarConcepts[
    Math.floor(Math.random() * grammarConcepts.length)
  ]

}
```

Meaning:

If something **needs review → practice it**
Otherwise → new grammar.
